#rev a list of numbers
li = [1, 2, 3, 4, 5, 6, 7,8,9]
left=0
right=len(li)-1
while left!=right:
  li[left],li[right]=li[right],li[left] 
  left+=1
  right-=1
print(li)

#Write a program to check for a Palindrome.
a=123454321
rev=0
temp=a
while temp>0:
  d=temp%10
  rev=(rev*10)+d
  temp//=10
if a==rev:
  print ("pallindrome") 
else:
  print("not pallindrome")  

#check a string is pallindrome or not
a = "radar"
is_palindrome = True
left = 0
right = len(a) - 1
while left < right:
    if a[left] != a[right]:
        is_palindrome = False
        break 
    left += 1
    right -= 1

print(is_palindrome)

  else:
    freq[i]=1

  is_prime=True

#Find the second largest number in a list.

second=0
for n in li:
  if n>fir:
    second=fir
    fir=n
  elif n>second:
    second=n
